cmake_minimum_required(VERSION 3.15)
project(usbcam_view CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(usbcam_view usbcam_view.cpp)
if(WIN32)
  # FFMPEG_DIR = extracted FFmpeg shared build (include/ lib/ bin/), SDL2_DIR = extracted SDL2-devel VC folder
  target_compile_definitions(usbcam_view PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
  target_include_directories(usbcam_view PRIVATE ${FFMPEG_DIR}/include ${SDL2_DIR}/include)
  target_link_directories(usbcam_view PRIVATE ${FFMPEG_DIR}/lib ${SDL2_DIR}/lib/x64)
  target_link_libraries(usbcam_view PRIVATE avcodec avutil SDL2 ws2_32)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(DEPS REQUIRED libavcodec libavutil sdl2)
  target_include_directories(usbcam_view PRIVATE ${DEPS_INCLUDE_DIRS})
  target_link_libraries(usbcam_view PRIVATE ${DEPS_LIBRARIES} pthread)
endif()
