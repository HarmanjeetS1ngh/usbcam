cmake_minimum_required(VERSION 3.15)
project(usbcam_vcam CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT WIN32)
  message(FATAL_ERROR "usbcam_vcam is Windows only")
endif()
# FFMPEG_DIR = extracted FFmpeg shared build (include/ lib/ bin/)

# 1) the camera component: runs inside Windows' Frame Server, so it uses the static C runtime and needs no extra DLLs
add_library(usbcam_vcam_src SHARED usbcam_vcam_src.cpp usbcam_vcam_src.def)
set_property(TARGET usbcam_vcam_src PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
target_compile_definitions(usbcam_vcam_src PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
target_link_libraries(usbcam_vcam_src PRIVATE mfplat mf mfuuid strmiids uuid ole32 advapi32 user32)

# 2) the feeder program (phone -> decode -> shared memory, creates the virtual camera)
add_executable(usbcam_vcam usbcam_vcam.cpp)
target_compile_definitions(usbcam_vcam PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
target_include_directories(usbcam_vcam PRIVATE ${FFMPEG_DIR}/include)
target_link_directories(usbcam_vcam PRIVATE ${FFMPEG_DIR}/lib)
target_link_libraries(usbcam_vcam PRIVATE avcodec avutil ws2_32 mfplat mf mfuuid mfsensorgroup strmiids uuid ole32 pdh gdi32 user32 advapi32)

# 3) the UsbCam app: window with live preview + controls + the virtual camera in one program (needs SDL2 and Dear ImGui)
if(IMGUI_DIR AND SDL2_DIR)
  enable_language(RC)
  # version shown in the update check: build = Actions run number (the release is tagged desktop-<build>), version = appVersionName
  if(NOT UCV_BUILD)
    set(UCV_BUILD 0)
  endif()
  if(NOT UCV_VERSION)
    set(UCV_VERSION "dev")
  endif()
  configure_file(usbcam_ver.h.in ${CMAKE_CURRENT_BINARY_DIR}/usbcam_ver.h @ONLY)
  add_executable(UsbCam WIN32 usbcam_vcam.cpp usbcam_gui.inc usbcam_update.inc usbcam_icon.h usbcam.rc
    ${IMGUI_DIR}/imgui.cpp ${IMGUI_DIR}/imgui_draw.cpp ${IMGUI_DIR}/imgui_tables.cpp ${IMGUI_DIR}/imgui_widgets.cpp
    ${IMGUI_DIR}/backends/imgui_impl_sdl2.cpp ${IMGUI_DIR}/backends/imgui_impl_sdlrenderer2.cpp)
  target_compile_definitions(UsbCam PRIVATE UCV_GUI _CRT_SECURE_NO_WARNINGS NOMINMAX)
  target_include_directories(UsbCam PRIVATE ${FFMPEG_DIR}/include ${SDL2_DIR}/include ${IMGUI_DIR} ${IMGUI_DIR}/backends ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR})
  target_link_directories(UsbCam PRIVATE ${FFMPEG_DIR}/lib ${SDL2_DIR}/lib/x64)
  target_link_libraries(UsbCam PRIVATE avcodec avutil SDL2 ws2_32 mfplat mf mfuuid mfsensorgroup strmiids uuid ole32 pdh gdi32 user32 advapi32 shell32 winhttp bcrypt)
  target_link_options(UsbCam PRIVATE /ENTRY:mainCRTStartup)
endif()
